```haskell
module Hyperdrive.Semantics.Field
  ( -- * Field names
    FieldName
  , mkFieldName
  , fieldNameBytes
    -- * Field values
  , FieldValue
  , FieldValueError(..)
  , mkFieldValue
  , fieldValueBytes
  , combineValues

  , Fields
  , emptyFields
  , fieldsFromList
  , fieldsToList
  , appendField
  , lookupSingleton
  , RepeatedField(..)
  , lookupFirst
  , lookupAll
    -- * Common field names
  , hConnection
  , hContentLength
  , hContentType
  , hHost
  , hSetCookie
  , hTransferEncoding
  ) where

-- | Validate a field name. Because field names are case-insensitive,
-- the name is normalized to lower case, which is also the form that
-- HTTP/2 and HTTP/3 require on the wire.
mkFieldName :: ByteString -> Maybe FieldName

-- | the field name, in lower case
fieldNameBytes :: FieldName -> ByteString

hConnection, hContentLength, hContentType, hHost, hSetCookie, hTransferEncoding :: FieldName

data FieldValueError
  = ContainsCRLFNUL -- ^ the value contains CR, LF or NUL
  | ContainsCTL     -- ^ the value contains some other control character
  deriving (Eq, Show)

-- | Validate a field value as received from the wire.
--
-- Leading and trailing whitespace is removed. Values containing CR,
-- LF or NUL are rejected rather than repaired. Values containing any
-- other control character are also rejected: we do not take up the
-- option to retain them.
--
-- RFC 9110 requires rejecting the whole /message/, not just the field
-- line, so a 'Left' must lead to the request being refused (typically
-- with 400 Bad Request). Dropping the field line and carrying on is
-- neither of the two behaviours the RFC allows.
mkFieldValue :: ByteString -> Either FieldValueError FieldValue

stripOWS :: ByteString -> ByteString

fieldValueBytes :: FieldValue -> ByteString

-- | Combine the values of several field lines with the same name into
-- a single value. Returns 'Nothing' for an empty list.
--
-- This is only meaningful for fields whose definition allows a
-- comma-separated list. Notably, @Set-Cookie@ does not.
combineValues :: [FieldValue] -> Maybe FieldValue

emptyFields :: Fields

fieldsFromList :: [(FieldName, FieldValue)] -> Fields

-- | the field lines, in order
fieldsToList :: Fields -> [(FieldName, FieldValue)]

-- | add a field line after all existing field lines. /O(1)/
appendField :: FieldName -> FieldValue -> Fields -> Fields

-- | A singleton field appeared on more than one field line.
newtype RepeatedField = RepeatedField FieldName
  deriving (Eq, Show)

-- | The value of a singleton field: one that may appear at most once,
-- such as @Host@, @Content-Type@ or @Authorization@. More than one
-- field line with the name is an error rather than a choice.
--
-- This is the lookup to use for any field that is not list-based.
-- Taking the first or the last of several lines is what makes duplicate
-- fields dangerous: a component in front of the server, such as a
-- proxy, cache or firewall, may make the opposite choice, and then the
-- two act on different values.
--
-- This only detects repeated field lines. A single line carrying
-- several members (@Content-Type: text/plain, text/html@) can only be
-- detected by the parser for that field, since some singleton fields,
-- such as @Date@, legitimately contain commas. /O(n)/
lookupSingleton :: FieldName -> Fields -> Either RepeatedField (Maybe FieldValue)

-- | The value of the first field line with this name, ignoring any
-- others. For singleton fields use 'lookupSingleton' instead: silently
-- picking one of several values is unsafe. /O(n)/
lookupFirst :: FieldName -> Fields -> Maybe FieldValue

-- | the values of every field line with this name, in order. /O(n)/
lookupAll :: FieldName -> Fields -> [FieldValue]
```

### FieldSyntax.hs

```haskell
-- | Common syntax used inside field values (RFC 9110, Section 5.6):
-- lists, quoted strings, comments and parameters.
module Hyperdrive.Semantics.FieldSyntax
  ( FieldSyntaxError(..)
    -- * Lists
  , Comments(..)
  , splitList
  , fieldListElements
    -- * Quoted strings
  , parseQuotedString
  , quoteString
    -- * Parameters
  , parseParameters
  , lookupParameter
  ) where

data FieldSyntaxError
  = UnterminatedQuotedString
  | InvalidQuotedString      -- ^ a character that is not allowed in a quoted-string
  | UnterminatedComment
  | UnexpectedCloseParen     -- ^ a ")" with no matching "("
  | InvalidParameter
  | DuplicateParameter ByteString -- ^ the (lower case) name of a parameter given more than once
  deriving (Eq, Show)

-- | Whether the elements of a list may contain comments. Comments are
-- only allowed in fields whose definition says so, such as @Via@; in
-- other fields "(" and ")" are ordinary characters.
data Comments = NoComments | AllowComments
  deriving (Eq, Show)

-- | Split a list-based field value into its elements.
--
-- Commas inside a quoted-string, or inside a comment when comments are
-- allowed, do not separate elements. Whitespace around each element is
-- removed and empty elements are dropped. Any number of empty elements
-- is accepted: they are skipped in linear time without allocation, so
-- the only limit needed is the one on the size of the field.
--
-- The result may be empty. Whether that is allowed depends on the
-- field: @1#element@ requires at least one element.
--
-- The elements themselves are not validated beyond their quoting;
-- that is up to the grammar of the particular field.
splitList :: Comments -> FieldValue -> Either FieldSyntaxError [ByteString]

-- | The elements of a list-based field, from every field line with
-- this name, in order.
--
-- Each field line is split on its own rather than after combining
-- them, so that a quote left open on one line cannot swallow the next.
-- For a well-formed message the result is the same.
fieldListElements :: Comments -> FieldName -> Fields -> Either FieldSyntaxError [ByteString]

isQDText :: Word8 -> Bool

isQuotedPairChar :: Word8 -> Bool

-- | Parse a quoted-string at the start of the input. Returns its value,
-- with each quoted-pair replaced by the octet it quotes, and the rest
-- of the input.
parseQuotedString :: ByteString -> Either FieldSyntaxError (ByteString, ByteString)

-- | Render a value as a quoted-string, escaping only DQUOTE and
-- backslash. Returns 'Nothing' if the value contains a control
-- character other than HTAB, which no quoted-string can carry.
quoteString :: ByteString -> Maybe ByteString

-- | Parse @parameters@: everything after the item they belong to, for
-- example the @; q=0.5@ of @gzip; q=0.5@. Names are returned in lower
-- case, since they are case-insensitive, and quoted values are
-- unquoted, since quoted and unquoted values are equivalent. Empty
-- parameters (@;;@) are skipped. Parameters are returned in order,
-- including any repeated names; use 'lookupParameter' to read them.
--
-- A value can be any printable text, such as @"../../etc/passwd"@.
-- Values that name files, like a @filename@ parameter, must be checked
-- with "Hyperdrive.Semantics.SafePath" before use.
parseParameters :: ByteString -> Either FieldSyntaxError [(ByteString, ByteString)]

-- | The value of the parameter with this name, which is compared
-- case-insensitively. A name given more than once is an error rather
-- than a choice between its values: as with repeated singleton fields,
-- different implementations choose differently, and an attacker can
-- send @charset=utf-8;charset=utf-7@ hoping that a filter checks one
-- and the application uses the other. /O(n)/
lookupParameter :: ByteString -> [(ByteString, ByteString)] -> Either FieldSyntaxError (Maybe ByteString)
```

### Internal.hs

```haskell
-- | Raw constructors for the semantic types.
--
-- The public modules only hand out values that satisfy each type's
-- invariant. These constructors bypass those checks and exist so that
-- protocol parsers which have /already/ validated their input do not
-- pay for a second pass. Anything built with them must satisfy the
-- invariant documented on the type.
module Hyperdrive.Semantics.Internal
  ( Method(..)
  , FieldName(..)
  , FieldValue(..)
  , Fields(..)
  , Status(..)
  , Scheme(..)
  , RegName(..)
  , IPvFuture(..)
  , Path(..)
  , Query(..)
  , SafeSegment(..)
  ) where

-- | Invariant: the bytes are a @token@.
newtype Method = Method ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes are a @token@ containing no upper case
-- letters.
newtype FieldName = FieldName ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes match @field-value@; in particular there is
-- no leading or trailing whitespace and no CR, LF or NUL.
newtype FieldValue = FieldValue ByteString
  deriving (Eq, Ord, Show)

-- | A field section (headers or trailers), in the order received.
--
-- A 'Seq' rather than a list, so that appending a field line is cheap.
-- With a list, building a section one field line at a time would be
-- quadratic in the number of field lines, which a client controls.
newtype Fields = Fields (Seq (FieldName, FieldValue))
  deriving (Eq, Show)

-- | Invariant: @100 <= code <= 599@.
newtype Status = Status Int
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes match @scheme@ and are in lower case.
newtype Scheme = Scheme ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes match @reg-name@, are not empty, and are in
-- normal form: no percent-encoded unreserved characters, letters in
-- lower case, and the hex digits of percent-encodings in upper case.
-- They do not match @IPv4address@.
newtype RegName = RegName ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes match @IPvFuture@ and are in lower case.
newtype IPvFuture = IPvFuture ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes match @path-abempty@.
newtype Path = Path ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes match @query@.
newtype Query = Query ByteString
  deriving (Eq, Ord, Show)

-- | Invariant: the bytes are a decoded path segment that is not @.@ or
-- @..@ and contains no "/" or NUL.
newtype SafeSegment = SafeSegment ByteString
  deriving (Eq, Ord, Show)
```

### Message.hs

```haskell
-- | Requests and responses (RFC 9110, Section 6)
--
-- These are the version-independent messages that applications see.
-- Each protocol layer (HTTP/1.1, HTTP/2, HTTP/3) turns what arrives on
-- the wire into a 'Request', and a 'Response' into what it sends, so an
-- application never sees which version of HTTP was used.
--
-- Fields that describe framing or the connection rather than the
-- message (@Content-Length@, @Transfer-Encoding@, @Connection@ and so
-- on) are not part of either: the protocol layer interprets them on the
-- way in and generates them on the way out, from the structure of the
-- message. So an application cannot, for example, send a
-- @Content-Length@ that disagrees with its content.
module Hyperdrive.Semantics.Message
  ( -- * Versions
    HttpVersion(..)
  , pattern HTTP_1_0
  , pattern HTTP_1_1
  , pattern HTTP_2
  , pattern HTTP_3
    -- * Requests
  , Request
  , RequestError(..)
  , mkRequest
  , requestVersion
  , requestMethod
  , requestTarget
  , requestHeaders
  , requestBody
  , RequestBody(..)
  , IncompleteBody(..)
  , bytesBody
    -- * Responses
  , Response
  , ResponseError(..)
  , mkResponse
  , responseStatus
  , responseHeaders
  , responseContent
  , Content(..)
  , FileRange
  , mkFileRange
  , rangeOffset
  , rangeLength
  , checkResponseFor
  , checkTrailers
    -- * Fields owned by the protocol layer
  , isProtocolField
  ) where

-- | An HTTP version, major and minor. Part of a request's control data:
-- some requirements depend on it, such as never sending a 1xx response
-- to an HTTP/1.0 client.
data HttpVersion = HttpVersion !Word8 !Word8
  deriving (Eq, Ord, Show)

pattern HTTP_1_0, HTTP_1_1, HTTP_2, HTTP_3 :: HttpVersion
pattern HTTP_1_0 = HttpVersion 1 0
pattern HTTP_1_1 = HttpVersion 1 1
pattern HTTP_2   = HttpVersion 2 0
pattern HTTP_3   = HttpVersion 3 0

-- | Fields that describe the framing of a message or the connection it
-- travels on, rather than the message itself: @Connection@,
-- @Keep-Alive@, @Proxy-Connection@, @TE@, @Transfer-Encoding@,
-- @Upgrade@ and @Content-Length@. They are interpreted and generated
-- by the protocol layer, and never appear in a 'Request' or 'Response'.
isProtocolField :: FieldName -> Bool

protocolFields :: [ByteString]

firstProtocolField :: Fields -> Maybe FieldName

-- | The content of a request did not arrive in full: the client
-- disconnected or reset the stream, or sent less or more than it
-- declared. Thrown by 'readChunk' and 'readTrailers'.
data IncompleteBody = IncompleteBody
  deriving (Eq, Show)

-- | The content of a request, read as it arrives. A body has a single
-- reader: reading it from more than one thread at a time is an error.
--
-- The guarantees every protocol layer gives:
--
-- * 'readChunk' returns an empty chunk only after the whole content has
--   arrived. If it does not arrive in full, 'readChunk' throws
--   'IncompleteBody', never an empty chunk, so a truncated upload is
--   never mistaken for a complete one.
--
-- * If 'bodyLength' is @Just n@, exactly @n@ bytes are delivered, or
--   'IncompleteBody' is thrown.
data RequestBody = RequestBody
  { -- | The length the client declared in advance, if it did. It comes
    -- from the client, so it is not a reason to allocate that much
    -- memory: compare it with your own limit first.
    bodyLength   :: !(Maybe Word64)
    -- | The next chunk of content. An empty chunk means the whole
    -- content has arrived, and every later call also returns an empty
    -- chunk.
  , readChunk    :: IO ByteString
    -- | The trailer section. Any content not yet read is read and
    -- discarded first, which can throw 'IncompleteBody'. Trailers are
    -- never merged into the header section (RFC 9110 Section 6.5.2).
  , readTrailers :: IO Fields
  }

-- | A body consisting of the given bytes, with no trailers. For tests,
-- and for code that builds requests itself.
bytesBody :: ByteString -> IO RequestBody

-- | A request, as an application sees it.
--
-- Invariants: the form of the target is valid for the method, an empty
-- path has its normal form, and the header fields contain no protocol
-- fields ('isProtocolField') and no @Host@. The authority the request
-- is for is in the target and nowhere else, so routing and the
-- application cannot act on two different hosts.
data Request = Request
  { requestVersion :: !HttpVersion
  , requestMethod  :: !Method
  , requestTarget  :: !Target
  , requestHeaders :: !Fields
  , requestBody    :: !RequestBody
  }

data RequestError
  = TargetNotValidForMethod     -- ^ the CONNECT or asterisk form, used with another method
  | RequestProtocolField !FieldName -- ^ a field the protocol layer should have removed
  deriving (Eq, Show)

-- | Build a request. The protocol layer has already reconstructed the
-- target and interpreted the protocol fields, expressing what they say
-- as the version, the target and the body. @Host@ (or HTTP/2 and
-- HTTP/3's @:authority@) has been folded into the target, and is
-- rejected here: leaving it in the fields would let an application
-- read a host that routing did not use, as when an absolute-form
-- target and @Host@ disagree.
mkRequest :: HttpVersion -> Method -> Target -> Fields -> RequestBody -> Either RequestError Request

-- | A range of bytes in a file: the offset of the first byte and the
-- number of bytes. Invariant: the end does not overflow.
data FileRange = FileRange { rangeOffset :: !Word64, rangeLength :: !Word64 }
  deriving (Eq, Show)

-- | A range from an offset and a length, or 'Nothing' if its end would
-- overflow. Whether the range lies within the file can only be checked
-- when it is sent, since the file can change in the meantime.
mkFileRange :: Word64 -> Word64 -> Maybe FileRange

-- | The content of a response.
--
-- 'ContentNone' (no content at all, as for 204) is different from empty
-- content (@'ContentBytes' ""@, content of zero length).
data Content
  = ContentNone
  | ContentBytes !ByteString
    -- | Part or all of a file, so that the IO layer can send it without
    -- copying it through user space. The path is chosen by the server,
    -- so any part that came from a request must have been checked with
    -- "Hyperdrive.Semantics.SafePath". The protocol layer sends exactly
    -- the length it announced: if the file has changed size by the time
    -- it is read, or the range lies outside it, the response is aborted
    -- rather than sending more or less.
  | ContentFile !FilePath !(Maybe FileRange)
    -- | Content produced as it is sent. The function is given a way to
    -- write a chunk and a way to flush what has been written, and
    -- returns the trailer section ('checkTrailers') once it has written
    -- all the content. If a length is given, exactly that many bytes
    -- must be written.
    --
    -- If the function throws, or writes more or less than the length it
    -- gave, the protocol layer aborts the response: it closes the
    -- connection or resets the stream, and never ends the content
    -- normally, so that a truncated response cannot pass for a
    -- complete one. Bytes beyond the declared length are never sent,
    -- since on an HTTP/1.1 connection a client would take them for the
    -- start of the next response. 'write' throws if the client has gone.
  | ContentStream !(Maybe Word64) ((ByteString -> IO ()) -> IO () -> IO Fields)

-- | A final response, as an application produces it.
--
-- Invariants: the status is not 1xx, there is no content exactly when
-- the status says there cannot be, and the header fields contain no
-- protocol fields ('isProtocolField').
data Response = Response
  { responseStatus  :: !Status
  , responseHeaders :: !Fields
  , responseContent :: !Content
  }

data ResponseError
  = InterimStatus                     -- ^ a 1xx status, which is not a final response
  | ContentNotAllowed                 -- ^ content with a 204, 205 or 304 status
  | ContentRequired                   -- ^ no content with any other status
  | ResponseProtocolField !FieldName  -- ^ a field only the protocol layer may generate
  | RepeatedSingleton !FieldName      -- ^ a field that may only appear once, appearing more than once
  | TunnelNotSupported                -- ^ a 2xx response to CONNECT, which would open a tunnel
  deriving (Eq, Show)

-- | Response fields whose grammar allows a single value, so that a
-- second field line could only make clients and caches disagree about
-- which one applies. From RFC 9110: Content-Type, Content-Range,
-- Content-Location, Location, ETag, Last-Modified, Retry-After, Date and
-- Server; from RFC 9111: Age and Expires.
singletonResponseFields :: [FieldName]

-- | Build a final response. Interim (1xx) responses are not 'Response's.
--
-- A response to HEAD is built as it would be for GET, content and all:
-- the protocol layer sends the fields that describe the content, such
-- as its length, and not the content itself.
mkResponse :: Status -> Fields -> Content -> Either ResponseError Response

-- | Check a response against the request it answers. The protocol layer
-- calls this before sending any response.
--
-- A 2xx response to CONNECT is rejected: a client takes it to mean the
-- connection is now a tunnel, and would treat whatever follows as
-- tunnelled data rather than HTTP. Tunnels are not supported yet.
checkResponseFor :: Request -> Response -> Either ResponseError Response

-- | Check the trailer section a 'ContentStream' returned, rejecting the
-- first field that cannot be a trailer.
--
-- RFC 9110 only lets a sender generate a trailer field whose definition
-- permits it, which is the application's responsibility. This catches
-- the fields that clearly cannot be trailers: the protocol fields, and
-- fields that describe framing, routing, authentication, request
-- modifiers, response controls or content format. Fields that RFC 9110
-- explicitly allows in trailers, such as @ETag@ and
-- @Authentication-Info@, are accepted.
checkTrailers :: Fields -> Either FieldName Fields

notTrailerFields :: [ByteString]
```

### SafePath.hs

```haskell
-- | Path segments that are safe to use as file or directory names.
--
-- 'Hyperdrive.Semantics.URI.pathSegments' decodes a path faithfully, so
-- a segment may be @..@, or may contain a "/" that was sent encoded as
-- @%2F@. Joining such segments into a file system path lets a request
-- climb out of the directory it was meant to stay in:
--
-- > /static/%2E%2E%2Fsecret  ==>  ["static", "../secret"]
--
-- A 'SafeSegment' can only be built from a segment that cannot do
-- that, so a sequence of them can be joined with "/" to name a file
-- beneath some root.
--
-- These checks are enough for POSIX file systems. Windows gives
-- special meaning to more names and characters, such as @\\@, @:@ and
-- device names like @aux@; code serving files from Windows must also
-- reject those.
module Hyperdrive.Semantics.SafePath
  ( SafeSegment
  , UnsafeSegment(..)
  , mkSafeSegment
  , safeSegmentBytes
  , safePathSegments
  ) where

data UnsafeSegment
  = DotSegment       -- ^ the segment was @.@ or @..@
  | ContainsSlash    -- ^ the segment contained "/", which was sent as @%2F@
  | ContainsNUL      -- ^ the segment contained NUL, which was sent as @%00@
  deriving (Eq, Show)

-- | Check a decoded segment. An empty segment is allowed: it names
-- nothing, and joining it produces "//", which file systems treat as
-- "/".
mkSafeSegment :: ByteString -> Either UnsafeSegment SafeSegment

safeSegmentBytes :: SafeSegment -> ByteString

-- | Decode a path into segments that are all safe to use as file names.
safePathSegments :: Path -> Either UnsafeSegment [SafeSegment]
```

### Target.hs

```haskell
-- | The target of a request (RFC 9110, Section 7.1)
--
-- Each protocol version reconstructs the target from different parts
-- of the message: HTTP/1.1 from the request-target and @Host@, HTTP/2
-- and HTTP/3 from the @:scheme@, @:authority@ and @:path@ pseudo-header
-- fields. This module is what they reconstruct it into.
module Hyperdrive.Semantics.Target
  ( Target(..)
  , connectTarget
  , validForMethod
  , normalizeEmptyPath
  ) where

data Target
  = -- | an ordinary target URI. The fragment, if any, is never part of
    -- the target.
    TargetURI !Scheme !Authority !Path !(Maybe Query)
    -- | the tunnel destination of a CONNECT request
  | TargetConnect !Host !Word16
    -- | @*@: the server as a whole, rather than a resource. Only used
    -- with OPTIONS.
  | TargetAsterisk
  deriving (Eq, Ord, Show)

-- | The target of a CONNECT request. There is no default port, so the
-- port must be present, and port 0 cannot be connected to.
connectTarget :: Authority -> Maybe Target

-- | The CONNECT and asterisk forms may only be used with CONNECT and
-- OPTIONS respectively. The reverse does not hold: an extended CONNECT
-- (RFC 8441, RFC 9220) has an ordinary target URI.
validForMethod :: Method -> Target -> Bool

-- | Give an empty path its normal form of "/", unless the method is
-- OPTIONS.
--
-- This is the only part of the scheme-based normalization of RFC 9110
-- Section 4.2.3 that is done here. In particular a default port is
-- kept, so @http://h:80/@ and @http://h/@ are different 'Target's, and
-- percent-encodings in the path and query are kept as received.
normalizeEmptyPath :: Method -> Target -> Target
```

```haskell
-- | The subset of URI syntax (RFC 3986) used by HTTP request targets.
--
-- Only absolute components are handled here: scheme, authority, path
-- and query. Relative references and reference resolution are not
-- needed by a server and are not implemented.
module Hyperdrive.Semantics.URI
  ( -- * Scheme
    Scheme

  , schemeBytes
  , pattern HTTP
  , pattern HTTPS
    -- * Authority
  , Authority(..)
  , AuthorityError(..)

  , renderAuthority

  , RegName

  , IPv6(..)
  , IPvFuture
  , ipvFutureBytes

    -- * Query
  , Query
  , mkQuery
  , queryBytes
    -- * Percent-encoding
  , percentDecode

-- | Validate a scheme, normalizing it to lower case.
mkScheme :: ByteString -> Maybe Scheme

schemeBytes :: Scheme -> ByteString

pattern HTTP, HTTPS :: Scheme
pattern HTTP  = Scheme "http"
pattern HTTPS = Scheme "https"

-- | A host, in normal form.
--
-- Only the dotted-decimal form of four octets is an IPv4 address, so
-- names like @1234@, @1.2.3@ or @0x7f.0.0.1@ are registered names.
-- System resolvers such as @getaddrinfo@ and @inet_aton@ may
-- nevertheless treat them as IPv4 addresses. Code that allows or denies
-- access by address must therefore resolve a registered name and check
-- the numeric result, never compare against the name's text; such names
-- can be rejected outright with 'isNumericLookalike'. To decide whether
-- two hosts are the same server, compare 'canonicalHost's, not 'Host's.
data Host
  = HostRegName   !RegName
  | HostIPv4      !Word32
  | HostIPv6      !IPv6
  | HostIPvFuture !IPvFuture
  deriving (Eq, Ord, Show)

-- | A 128-bit IPv6 address, most significant half first
data IPv6 = IPv6 !Word64 !Word64
  deriving (Eq, Ord, Show)

-- | The authority of a target URI. The port is 'Nothing' when it was
-- empty or not given.
data Authority = Authority
  { authorityHost :: !Host
  , authorityPort :: !(Maybe Word16)
  }
  deriving (Eq, Ord, Show)

data AuthorityError
  = AuthorityHasUserinfo
  | AuthorityEmptyHost
  | AuthorityInvalidHost
  | AuthorityInvalidPort
  deriving (Eq, Show)

-- | Parse @uri-host [ ":" port ]@. This is the syntax of the @Host@
-- header field, the @:authority@ pseudo-header field and the authority
-- of an http(s) URI once its userinfo is rejected.
--
-- Userinfo is treated as an error. An empty host is rejected, since
-- an empty host is invalid for both http and https. Ports above 65535
-- are rejected, since they cannot identify a TCP or UDP port.
parseAuthority :: ByteString -> Either AuthorityError Authority

-- | Split at the ":" that introduces the port. A ":" can only appear
-- in the host inside an IP-literal.
splitHostPort :: ByteString -> Either AuthorityError (ByteString, Maybe ByteString)

parseHost :: ByteString -> Either AuthorityError Host

isRegName :: ByteString -> Bool

-- | Percent-encoded unreserved characters are decoded, letters are made
-- lower case, and the hex digits of the remaining percent-encodings
-- are made upper case.
normalizeRegName :: ByteString -> ByteString

isIPvFuture :: ByteString -> Bool

decOctet :: ByteString -> Maybe Word8

-- | The nine alternatives come down to: without "::" there must be
-- exactly eight 16-bit pieces; with a single "::" there may be at most
-- seven. In both cases an IPv4 address may only appear at the end,
-- where it counts as two pieces.
parseIPv6 :: ByteString -> Maybe IPv6

-- recommended by RFC 5952.
renderHost :: Host -> ByteString

-- | RFC 5952: lower case hex, no leading zeros, and the longest run of
-- two or more zero pieces (the first, if there is a tie) replaced by
-- "::". IPv4-mapped addresses (@::ffff:0:0/96@, RFC 4291) end in dotted
-- decimal. No other prefix does: not RFC 4291's deprecated
-- IPv4-compatible prefix (@::/96@), which covers @::1@, nor the
-- obsolete IPv4-translated prefix of RFC 2765 (@::ffff:0:0:0/96@), nor
-- RFC 6052's @64:ff9b::/96@.
renderIPv6 :: IPv6 -> ByteString

longestZeroRun :: [Word16] -> Maybe (Int, Int)

-- | A host in a form suitable for deciding whether two hosts are the
-- same server, for example when matching virtual hosts or checking an
-- allow-list. Two spellings that the network treats as the same are
-- mapped to one:
--
-- * an IPv4-mapped IPv6 address (@[::ffff:127.0.0.1]@) becomes the IPv4
--   address it maps (@127.0.0.1@), which is what a dual-stack socket
--   connects to;
--
-- * a single trailing dot is removed from a registered name, so the
--   fully-qualified @example.com.@ matches @example.com@. If the result
--   is an IPv4 address (@1.2.3.4.@), it becomes one. A name ending in
--   more than one dot has an empty label and is not the same name, so
--   it is left alone; this also keeps @canonicalHost@ idempotent.
--
-- This is not a normal form for URIs: @example.com.@ and @example.com@
-- are different URIs, and render differently. Use it for comparisons,
-- not for output.
--
-- Numeric names that system resolvers may read as IPv4 addresses
-- (@2130706433@, @0x7f.1@) are not mapped, since resolvers differ.
-- Reject them with 'isNumericLookalike' instead.
canonicalHost :: Host -> Host

-- | remove exactly one trailing dot, if there is one and it is not
-- preceded by another
stripTrailingDot :: ByteString -> Maybe ByteString

-- | The port a request is for: the one given, or else the default for
-- http (80) or https (443). 'Nothing' for another scheme with no port.
effectivePort :: Scheme -> Maybe Word16 -> Maybe Word16

-- | A key for comparing the authorities of two targets with the same
-- scheme: @http://example.com:80@ and @http://Example.COM.@ have the
-- same key.
authorityKey :: Scheme -> Authority -> (Host, Maybe Word16)

-- | Is this registered name numeric in its last label, like
-- @2130706433@, @127.1@, @0177.0.0.1@ or @0x7f.0.0.1@?
--
-- By the URI grammar these are names rather than IPv4 addresses, but
-- system resolvers such as @inet_aton@ and @getaddrinfo@ may read them
-- as addresses, in decimal, octal or hexadecimal. No real host name has
-- a numeric top-level label (RFC 1123, Section 2.1), so a server can
-- safely reject such names, and should do so before applying any
-- allow-list or deny-list to them. A single trailing dot is ignored.
isNumericLookalike :: RegName -> Bool

-- | Is this registered name a host name as DNS defines it: labels of
-- 1 to 63 letters, digits and hyphens that start and end with a letter
-- or digit, at most 253 characters in all, with an optional single
-- trailing dot, and not numeric in its last label ('isNumericLookalike')?
--
-- Browsers only ever send such names: internationalized names are sent
-- in their ASCII (punycode) form. Anything else that RFC 3986 allows in
-- a registered name, such as percent-encoding (@ho%00st@, @ho%40st@),
-- sub-delims or underscores, is not a name a client will legitimately
-- ask for, and can be dangerous to anything that decodes or
-- reinterprets it. A server can therefore require this of every @Host@
-- (docs/design-notes.md, P10).
isDnsHostName :: RegName -> Bool

renderAuthority :: Authority -> ByteString

-- | Validate a path. The path is kept exactly as received, including
-- its percent-encoding, because decoding can change its meaning (an
-- encoded "/" is not a segment separator).
mkPath :: ByteString -> Maybe Path

pathBytes :: Path -> ByteString

data PathSegmentError
  = SegmentContainsNUL -- ^ a segment contained @%00@
  deriving (Eq, Show)

-- | Split a path into segments, then percent-decode each one.
--
-- > pathSegments "/"            == Right []
-- > pathSegments "/a/b%2Fc/"    == Right ["a", "b/c", ""]
-- > pathSegments "/a/%2e%2e/b"  == Right ["a", "..", "b"]
-- > pathSegments "/a%00"        == Left SegmentContainsNUL
--
-- This is a faithful decoding: a segment may be @.@ or @..@, and an
-- encoded "/" becomes part of a segment rather than a separator. Both
-- are legitimate data to an application that is not using the path to
-- name files. Anything that does map segments to files or other
-- storage must use "Hyperdrive.Semantics.SafePath" instead.
--
-- A decoded segment may also contain any control character other than
-- NUL, including CR, LF and ESC. Anything that writes segments to a
-- log or terminal must escape them. (Response headers are protected:
-- a 'Hyperdrive.Semantics.Field.FieldValue' cannot contain CR or LF.)
--
-- Routing and access control should both use these segments. Deciding
-- access on the raw 'Path' and routing on the segments, or the other
-- way round, lets differently spelled requests (@/%61dmin@, @/./admin@,
-- @//admin@) pass a check meant for the path they reach.
pathSegments :: Path -> Either PathSegmentError [ByteString]

-- | Validate a query, keeping it exactly as received.
mkQuery :: ByteString -> Maybe Query

queryBytes :: Query -> ByteString

-- | Every octet is either allowed by the predicate or is the start of
-- a @pct-encoded@ triplet.
allEncoded :: (Word8 -> Bool) -> ByteString -> Bool

-- | Decode percent-encoded octets. Returns 'Nothing' if a "%" is not
-- followed by two hex digits.
percentDecode :: ByteString -> Maybe ByteString

hexValue :: Word8 -> Word8

toLower :: Word8 -> Word8

toUpper :: Word8 -> Word8
```
